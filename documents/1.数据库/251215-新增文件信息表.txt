-- ########################### 文件管理表 ###########################
-- 1. 文件信息表（file_info）
CREATE TABLE file_info (
  `id` bigint(20) NOT NULL AUTO_INCREMENT COMMENT '文件ID',
  `original_filename` varchar(255) NOT NULL COMMENT '原始文件名',
  `filename` varchar(255) NOT NULL COMMENT '存储的文件名',
  `file_path` varchar(500) NOT NULL COMMENT '文件存储路径',
  `size` bigint(20) NOT NULL COMMENT '文件大小（字节）',
  `content_type` varchar(100) DEFAULT NULL COMMENT '文件类型',
  `download_url` varchar(500) DEFAULT NULL COMMENT '下载URL',
  `preview_url` varchar(500) DEFAULT NULL COMMENT '预览URL',
  `upload_time` datetime NOT NULL COMMENT '上传时间',
  `upload_user_id` bigint(20) DEFAULT NULL COMMENT '上传用户ID',
  `upload_username` varchar(100) DEFAULT NULL COMMENT '上传用户名',
  `md5` varchar(32) DEFAULT NULL COMMENT '文件MD5值',
  `sha256` varchar(64) DEFAULT NULL COMMENT '文件SHA256值',
  `description` varchar(500) DEFAULT NULL COMMENT '文件描述',
  `category` varchar(100) DEFAULT NULL COMMENT '文件分类',
  `tags` varchar(500) DEFAULT NULL COMMENT '文件标签',
  `is_public` tinyint(1) DEFAULT '1' COMMENT '是否公开（0=私有，1=公开）',
  `download_count` int(11) DEFAULT '0' COMMENT '下载次数',
  `status` tinyint(1) DEFAULT '1' COMMENT '状态（0=禁用，1=正常）',
  `deleted` tinyint(1) DEFAULT '0' COMMENT '是否删除（0=未删除，1=已删除）',
  `create_time` datetime DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_time` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_filename` (`filename`),
  KEY `idx_upload_user` (`upload_user_id`),
  KEY `idx_category` (`category`),
  KEY `idx_upload_time` (`upload_time`),
  KEY `idx_md5` (`md5`),
  KEY `idx_status` (`status`)
);
